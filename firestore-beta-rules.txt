rules_version = '2';
service cloud.firestore {
 match /databases/{database}/documents {
 function verified(){return request.auth != null && request.auth.token.email_verified == true;}
 function member(room){return verified() && request.auth.uid in get(/databases/$(database)/documents/mwChats/$(room)).data.members;}
 match /mwUsernames/{name} {
 allow get: if verified();
 allow create: if verified() && name.matches('^[a-z0-9_]{4,24}$') && request.resource.data.keys().hasOnly(['uid']) && request.resource.data.uid == request.auth.uid;
 }
 match /mwChats/{room} {
 allow get: if verified() && (!exists(/databases/$(database)/documents/mwChats/$(room)) || request.auth.uid in resource.data.members);
 allow create: if verified() && request.resource.data.keys().hasOnly(['members']) && request.resource.data.members is list && request.resource.data.members.size()==2 && request.resource.data.members[0] is string && request.resource.data.members[1] is string && request.resource.data.members[0]!=request.resource.data.members[1] && request.auth.uid in request.resource.data.members;
 match /messages/{id} {
 allow read: if member(room);
 allow create: if member(room) && request.resource.data.keys().hasOnly(['sender','text','created']) && request.resource.data.sender==request.auth.uid && request.resource.data.text is string && request.resource.data.text.size()>0 && request.resource.data.text.size()<=1000 && request.resource.data.created==request.time;
 }
 }

// Merge these matches into existing match /databases/{database}/documents.
// Remove overlapping permissive rules. Test in emulator before publishing.
match /mwCustomerIds/{uid} {
 allow get: if request.auth != null && request.auth.token.email_verified == true && request.auth.uid == uid;
 allow list, update, delete: if false;
 allow create: if request.auth != null && request.auth.token.email_verified == true && request.auth.uid == uid
 && request.resource.data.keys().hasOnly(['number'])
 && request.resource.data.number is int
 && request.resource.data.number > 0
 && getAfter(/databases/$(database)/documents/mwCounters/customers).data.uid == uid
 && getAfter(/databases/$(database)/documents/mwCounters/customers).data.last == request.resource.data.number
 && (!exists(/databases/$(database)/documents/mwCounters/customers) ? request.resource.data.number == 1 : request.resource.data.number == get(/databases/$(database)/documents/mwCounters/customers).data.last + 1);
}
match /mwCounters/customers {
 allow get: if request.auth != null && request.auth.token.email_verified == true;
 allow list, delete: if false;
 allow create, update: if request.auth != null && request.auth.token.email_verified == true
 && request.resource.data.keys().hasOnly(['last','uid'])
 && request.resource.data.uid == request.auth.uid
 && request.resource.data.last is int
 && (!exists(/databases/$(database)/documents/mwCounters/customers) ? request.resource.data.last == 1 : request.resource.data.last == resource.data.last + 1)
 && !exists(/databases/$(database)/documents/mwCustomerIds/$(request.auth.uid))
 && getAfter(/databases/$(database)/documents/mwCustomerIds/$(request.auth.uid)).data.number == request.resource.data.last;
}
// Merge this match block inside match /databases/{database}/documents.
// Do not replace unrelated rules. Remove any overlapping broad allow rules.
match /creditBooks/{uid}/entries/{entry} {
 allow read: if request.auth != null && request.auth.uid == uid;
 allow create: if request.auth != null && request.auth.uid == uid
 && request.resource.data.keys().hasAll(['customer','phone','kind','paise','date','note','createdAt'])
 && request.resource.data.keys().hasOnly(['customer','phone','kind','paise','date','note','createdAt'])
 && request.resource.data.customer is string && request.resource.data.customer.size() > 0 && request.resource.data.customer.size() <= 100
 && request.resource.data.phone is string && request.resource.data.phone.matches('^[0-9]{10} {allow read,write: if false;}
 }
})
 && request.resource.data.kind in ['credit','payment']
 && request.resource.data.paise is int && request.resource.data.paise > 0 && request.resource.data.paise <= 1000000000
 && request.resource.data.date is string && request.resource.data.date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2} {allow read,write: if false;}
 }
})
 && request.resource.data.note is string && request.resource.data.note.size() <= 300
 && request.resource.data.createdAt == request.time;
 allow update, delete: if false;
}
 match /{document=**} {allow read,write: if false;}
 }
}
