// Merge inside match /databases/{database}/documents. Keep all existing rules.
// Private user-maintained ledger, not a verified financial record.
match /mwDebts/{uid}/loans/{loan} {
 function owner(){return request.auth != null && request.auth.uid == uid && request.auth.token.email_verified == true;}
 function valid(){let d=request.resource.data;return d.keys().hasAll(['name','amount','initial','taken','due','rate','desc','payments','updated']) && d.keys().hasOnly(['name','amount','initial','taken','due','rate','desc','payments','updated']) && d.name is string && d.name.size()>0 && d.name.size()<=100 && d.amount is int && d.amount>0 && d.amount<=10000000000 && d.initial is int && d.initial>=0 && d.initial<=d.amount && d.taken is string && d.taken.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') && d.due is string && (d.due=='' || d.due.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$')) && d.rate is number && d.rate>=0 && d.rate<=1000 && d.desc is string && d.desc.size()<=500 && d.payments is list && d.payments.size()<=500 && d.updated==request.time;}
 allow read: if owner();
 allow create: if owner() && valid() && request.resource.data.payments.size()==0;
 allow update: if owner() && valid() && (request.resource.data.payments==resource.data.payments || (request.resource.data.payments.size()==resource.data.payments.size()+1 && request.resource.data.payments==resource.data.payments.concat([request.resource.data.payments[resource.data.payments.size()]]) && request.resource.data.payments[resource.data.payments.size()].keys().hasOnly(['amount','date','note']) && request.resource.data.payments[resource.data.payments.size()].amount is int && request.resource.data.payments[resource.data.payments.size()].amount>0 && request.resource.data.payments[resource.data.payments.size()].amount<=10000000000 && request.resource.data.payments[resource.data.payments.size()].date is string && request.resource.data.payments[resource.data.payments.size()].date.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$') && request.resource.data.payments[resource.data.payments.size()].note is string && request.resource.data.payments[resource.data.payments.size()].note.size()<=200));
 allow delete: if false;
}
